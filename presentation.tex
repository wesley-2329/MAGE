% Slide 2: Motivation & Objectives
\begin{frame}{Motivation \& Objectives}
\begin{itemize}
    \item \textbf{Generalization Gap in AI Text Detection:} Supervised binary detectors perform well in closed labs but fail under real-world distribution shifts (unseen domains/models).
    \item \textbf{Closed-Set Attributor Limitations:} Existing supervised attributors force every machine-generated text into a known class (e.g., classifying a text as GPT or LLaMA even if it was generated by OPT).
    \item \textbf{Core Objective:} Implement a Stage-2 open-set rejection layer that models known generator distributions to detect and flag unseen generators as \textbf{UNKNOWN}.
\end{itemize}
\end{frame}

    \node (output) [below of=stage2, draw=DarkOrange, ultra thick] {Attribution (Known) \\ or Rejection (UNKNOWN)};
    
    \draw [arrow] (input) -- (stage1);
    \draw [arrow] (stage1) -- (route);
    \draw [arrow] (route) -- node[left, draw=none] {YES} (stage2);
    \draw [arrow] (stage2) -- (output);
    \end{tikzpicture}
    }
\end{column}
\begin{column}{0.5\textwidth}
    \textbf{Proposed OOD Rejection:}
    \begin{itemize}
        \item Classifier is trained on classes $A$ (GPT) and $B$ (LLaMA).
        \item Incoming test sample $C$ (OPT) is evaluated in a 768-D embedding space.
        \item Rejection layer computes the distance to known distributions.
        \item If distance $> \tau$ (validation threshold), sample is flagged as \textbf{UNKNOWN} to prevent false attribution.
    \end{itemize}
\end{column}
\end{columns}
\end{frame}

% Slide 6: Proposed Open-Set Rejection Layer
\begin{frame}{Proposed Open-Set Rejection Layer}
\begin{block}{1. Class Centroids & Regularized Class Covariance}
The mean embedding vector $\mu_c$ and class covariance $\Sigma_c$ for each known class $c$ in training set:
$$\mu_c = \frac{1}{N_c} \sum_{i=1}^{N_c} x_i^{(c)} \qquad \Sigma_c = \text{Cov}(X^{(c)}) + 10^{-4} \cdot I$$
\end{block}
\begin{block}{2. Minimum Mahalanobis Distance & Zero-Leakage Thresholding}
Measuring OOD score $D(x)$ relative to the covariance-scaled known class distributions \cite{mahalanobis1936generalized}:
$$D(x) = \min_{c \in \{0, 1\}} \sqrt{(x - \mu_c)^T \Sigma_c^{-1} (x - \mu_c)}$$
The threshold $\tau = 63.5972$ is selected at the \textbf{95th percentile} of the known validation set's distances. Unseen OPT is completely absent during training, validation, and threshold selection.
\end{block}
\end{frame}

\begin{frame}{Representation \& Distance Studies (Exps 1 \& 2)}
\centering
\resizebox{0.8\textwidth}{!}{
\begin{tabular}{lcccc}
\toprule
\textbf{Configuration} & \textbf{Unseen OPT Rejection} & \textbf{Known False Rejection} & \textbf{OOD AUROC} & \textbf{Threshold} \\
\midrule
\textbf{Baseline (CLS + Mahalanobis)} & \textbf{7.59\%} & \textbf{3.90\%} & \textbf{0.6298} & \textbf{63.5972} \\
\textbf{Experiment 1 (Mean Pooling)} & 1.61\% & 4.40\% & 0.5251 & 5.8443 \\
\textbf{Experiment 2 (Cosine Distance)} & 0.85\% & 3.70\% & 0.4898 & 0.0158 \\
\bottomrule
\end{tabular}
}
\begin{itemize}
    \item \textbf{Exp 1 Interpretation:} Fine-tuning focuses the attribution gradients entirely on the \texttt{[CLS]} token. Mean pooling averages across all tokens, causing semantic signatures to wash out style.
    \item \textbf{Exp 2 Interpretation:} Fine-tuning collapses embeddings into a collinear cone. Cosine distance (angular) fails to separate distributions without covariance scaling (Mahalanobis).
\end{itemize}
\end{frame}

% Slide 10: Conclusions, Limitations & References
\begin{frame}[allowframebreaks]{Conclusions, Limitations \& References}
\begin{columns}
\begin{column}{0.5\textwidth}
    \textbf{Limitations \& Future Work}
    \begin{itemize}
        \item \textbf{Collinearity:} Open-source models (LLaMA/OPT) overlap heavily.
        \item \textbf{Future Work:} Triplet/Contrastive loss (metric learning) to compress known clusters and expand OOD margin.
    \end{itemize}
\end{column}
\begin{column}{0.5\textwidth}
    \textbf{Key Conclusions}
    \begin{itemize}
        \item CLS pooling captures generator style features far better than Mean pooling.
        \item Covariance scaling (Mahalanobis) is essential to expand angular cone.
    \end{itemize}
\end{column}
\end{columns}
